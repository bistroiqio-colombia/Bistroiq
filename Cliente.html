let carritoBorrador = {}; // Productos seleccionados antes de enviar // Función auxiliar para slugs function obtenerIdSlug(nombre) { return nombre.toLowerCase() .replace(/á/g, 'a').replace(/é/g, 'e').replace(/í/g, 'i').replace(/ó/g, 'o').replace(/ú/g, 'u') .replace(/\s+/g, '-'); } // 1. Modificar Selección Local (+ / -) window.modificarSeleccion = function(nombre, precio, cambio) { if (!carritoBorrador[nombre]) { carritoBorrador[nombre] = { nombre, precio, cantidad: 0 }; } carritoBorrador[nombre].cantidad += cambio; const slug = obtenerIdSlug(nombre); const elementoCantidad = document.getElementById(`cant-${slug}`); if (carritoBorrador[nombre].cantidad <= 0) { delete carritoBorrador[nombre]; if (elementoCantidad) elementoCantidad.textContent = 0; } else { if (elementoCantidad) elementoCantidad.textContent = carritoBorrador[nombre].cantidad; } actualizarVistaCarrito(); }; // 2. Botón de Papelera para eliminar un plato por completo del carrito window.eliminarItemCarrito = function(nombre) { if (carritoBorrador[nombre]) { delete carritoBorrador[nombre]; const slug = obtenerIdSlug(nombre); const elementoCantidad = document.getElementById(`cant-${slug}`); if (elementoCantidad) elementoCantidad.textContent = 0; actualizarVistaCarrito(); } }; // 3. Actualizar la vista del resumen y la barra flotante en tiempo real function actualizarVistaCarrito() { const listaContainer = document.getElementById("cliente-pedido-lista"); const totalElement = document.getElementById("cliente-total"); const totalFlotante = document.getElementById("total-flotante"); const barra = document.getElementById("barra-flotante-envio"); const items = Object.values(carritoBorrador); let subtotalBorrador = 0; let cantidadTotal = 0; if (items.length === 0) { listaContainer.innerHTML = `
Mesa limpia. Selecciona tus platos en el menú.
`; totalElement.textContent = "$ 0 COP"; totalFlotante.textContent = "$ 0 COP"; barra.classList.add("translate-y-full"); return; } listaContainer.innerHTML = items.map(item => { subtotalBorrador += item.precio * item.cantidad; cantidadTotal += item.cantidad; return `
${item.nombre}
Cantidad: ${item.cantidad}
$ ${(item.precio * item.cantidad).toLocaleString("es-CO")} COP
`; }).join(''); totalElement.textContent = `$ ${subtotalBorrador.toLocaleString("es-CO")} COP`; totalFlotante.textContent = `$ ${subtotalBorrador.toLocaleString("es-CO")} COP`; barra.classList.remove("translate-y-full"); }
